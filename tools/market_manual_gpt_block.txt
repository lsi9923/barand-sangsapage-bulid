
    def open_manual_market_url():
        market_url = self.market_manual_url_field.text().strip()
        if not re.match(r"^https?://", market_url, flags=re.IGNORECASE):
            QMessageBox.information(self, "URL 확인", "쿠팡 수동등록 URL을 http:// 또는 https://부터 입력하세요.")
            return
        self._open_url_in_automation_browser(
            market_url,
            "쿠팡 수동등록 URL을 자동화 Chrome에 열었습니다.",
        )

    def send_coupang_manual_gpt_prompt():
        product = self._selected_market_product()
        if product is None:
            QMessageBox.information(self, "선택 필요", "완료폴더를 스캔한 뒤 ChatGPT로 보낼 상품을 선택하세요.")
            return
        market_url = self.market_manual_url_field.text().strip()
        if not re.match(r"^https?://", market_url, flags=re.IGNORECASE):
            QMessageBox.information(self, "URL 확인", "쿠팡 수동등록 URL을 http:// 또는 https://부터 입력하세요.")
            return
        if getattr(self, "market_gpt_running", False):
            QMessageBox.information(self, "전송 중", "이미 쿠팡 수동등록 프롬프트를 ChatGPT로 보내는 중입니다.")
            return

        self._persist_market_settings_silent()
        draft = self._write_market_draft_bundle(product)
        base_prompt = str(draft.get("coupang_chatgpt_prompt") or "").strip()
        if not base_prompt:
            QMessageBox.warning(self, "초안 없음", "선택 상품의 쿠팡 마켓 프롬프트를 만들지 못했습니다.")
            return
        coupang_payload = draft.get("coupang_payload") if isinstance(draft, dict) else {}
        seo_keywords = draft.get("seo_keywords") if isinstance(draft, dict) else {}
        pricing = {}
        if isinstance(coupang_payload, dict):
            draft_info = coupang_payload.get("_draft")
            if isinstance(draft_info, dict) and isinstance(draft_info.get("pricing"), dict):
                pricing = draft_info.get("pricing")
        sale_price = self._format_won(pricing.get("sale_price", 0))

        from urllib.parse import urlparse

        def compact_value(value, limit=3500):
            if value in (None, "", [], {}):
                return ""
            if isinstance(value, str):
                text = value
            else:
                try:
                    text = json.dumps(value, ensure_ascii=False, separators=(",", ":"))
                except Exception:
                    text = str(value)
            text = re.sub(r"\s+", " ", text).strip()
            return text if len(text) <= limit else text[:limit] + "..."

        def classify_url(value):
            parsed = urlparse(str(value or "").strip())
            host = parsed.netloc.lower().split(":", 1)[0]
            if host == "wing.coupang.com" or host.endswith(".wing.coupang.com"):
                role = "쿠팡 Wing 입력 화면(상품 사실 근거로 사용 금지)"
            elif host == "coupang.com" or host.endswith(".coupang.com"):
                role = "쿠팡 참고 상품 페이지(가격·재고·후기는 변동 가능)"
            elif host:
                role = "외부 공급/참고 상품 페이지(접근 및 로그인 여부 확인 필요)"
            else:
                role = "잘못된 URL"
            return {"url": str(value or "").strip(), "host": host, "role": role}

        source = product.source if isinstance(product.source, dict) else {}
        metadata = product.metadata if isinstance(product.metadata, dict) else {}
        source_url = str(metadata.get("url") or source.get("url") or "").strip()
        secondary_url = str(metadata.get("secondary_url") or source.get("secondary_url") or "").strip()
        source_payloads = metadata.get("source_payloads")
        payload_summaries = []
        if isinstance(source_payloads, list):
            for payload in source_payloads[:4]:
                if not isinstance(payload, dict):
                    continue
                payload_summaries.append({
                    "origin": compact_value(payload.get("origin"), 80),
                    "url": compact_value(payload.get("url"), 1000),
                    "usage": compact_value(payload.get("usage"), 300),
                    "image_only": bool(payload.get("image_only", False)),
                    "title": compact_value(payload.get("title"), 500),
                })
        payload_items = coupang_payload.get("items") if isinstance(coupang_payload, dict) else []
        item_summaries = []
        if isinstance(payload_items, list):
            for item in payload_items[:30]:
                if not isinstance(item, dict):
                    continue
                item_summaries.append({
                    "itemName": compact_value(item.get("itemName"), 300),
                    "salePrice": item.get("salePrice"),
                    "maximumBuyCount": item.get("maximumBuyCount"),
                    "outboundShippingTimeDay": item.get("outboundShippingTimeDay"),
                    "attributes": item.get("attributes") if isinstance(item.get("attributes"), list) else [],
                })
        url_checks = [classify_url(market_url)]
        for saved_url in (source_url, secondary_url):
            if saved_url and all(row.get("url") != saved_url for row in url_checks):
                url_checks.append(classify_url(saved_url))
        evidence = {
            "schema": "coupang-manual-evidence-v2",
            "product_code": product.code,
            "completed_folder": str(product.folder),
            "manual_url_check": url_checks[0],
            "saved_url_checks": url_checks[1:],
            "source": {
                "original_product_name": compact_value(product.product_name, 1000),
                "title": compact_value(source.get("title"), 1000),
                "category": compact_value(metadata.get("category") or source.get("category"), 500),
                "price_text": compact_value(metadata.get("price_text") or source.get("price_text"), 500),
                "options_text": compact_value(metadata.get("options_text") or source.get("options_text"), 1000),
                "option_rows": compact_value(source.get("option_rows"), 3500),
                "facts": compact_value(source.get("facts"), 3500),
                "page_text_excerpt": compact_value(source.get("text"), 3500),
                "source_payloads": payload_summaries,
            },
            "app_draft": {
                "sellerProductName": compact_value(coupang_payload.get("sellerProductName") if isinstance(coupang_payload, dict) else "", 500),
                "displayCategoryCode": coupang_payload.get("displayCategoryCode") if isinstance(coupang_payload, dict) else None,
                "sale_price": pricing.get("sale_price", 0),
                "pricing": pricing,
                "category_candidates": seo_keywords.get("category_candidates", []) if isinstance(seo_keywords, dict) else [],
                "search_tags": seo_keywords.get("coupang_search_tags", []) if isinstance(seo_keywords, dict) else [],
                "items": item_summaries,
            },
            "attachments": [path.name for path in product.thumbnail_paths[:5] if self._is_valid_image_file(path)],
            "evidence_priority": [
                "완료폴더 source.json/metadata.json의 명시값과 첨부 이미지",
                "앱이 생성한 쿠팡 payload 및 계산 판매가",
                "접근 가능한 참고 상품 URL",
                "모델 추론은 입력값으로 사용 금지",
            ],
        }
        evidence_json = json.dumps(evidence, ensure_ascii=False, indent=2)
        manual_prompt = f"""{base_prompt}

이번 요청은 상세페이지 제작이 아니라 쿠팡 Wing 수동 상품등록용 최종 입력표 작성이다.
사용자가 입력한 마켓 등록 URL: {market_url}
입력 URL 판정: {url_checks[0]['role']}
앱에서 계산한 쿠팡 판매가: {sale_price}

[완료폴더 및 URL 대조 근거 JSON]
{evidence_json}

반드시 지킬 기준:
- 근거 우선순위는 `완료폴더 명시값/첨부 이미지 > 앱 payload > 접근 가능한 참고 URL`이다. 서로 충돌하면 임의 선택하지 말고 충돌표에 모두 적는다.
- Wing URL은 입력 위치 확인용이지 상품 사실의 근거가 아니다. 참고 URL에 접근하지 못해도 보이는 척하거나 내용을 추측하지 않는다.
- 위 쿠팡 전용 초안과 수집된 사실만 사용하고, 확인되지 않은 인증번호·모델명·소재·치수·용량·중량·효능은 만들지 않는다.
- 일반 상품은 상품명을 '끄롱마제'로 시작하고 판매처명·최저가·무료배송 같은 광고 문구를 제거한다.
- 원본명이나 이미지에 타사 브랜드·차종·캐릭터·작품명·로고가 있으면 이를 숨기거나 '끄롱마제' 제품으로 바꾸지 않는다. 정품/판매권 근거가 없으면 `등록 보류 - 권리 확인 필요`로 판정한다.
- 판매가는 앱 계산값을 임의로 다시 계산하거나 바꾸지 않는다.
- 옵션명과 옵션값은 source의 option_rows/options_text 또는 첨부 이미지로 확인되는 값만 사용한다. '옵션', '상세 조건 확인', 번역이 불명확한 문장을 등록값으로 쓰지 않는다.
- 가격·옵션·카테고리·필수 속성·인증정보가 비어 있거나 서로 충돌하면 값을 꾸며내지 말고 `확인 필요`로 남긴다.
- 중국어/영어 상품명은 상품 종류가 이미지와 근거에서 일치할 때만 자연스러운 한국어 SEO 명칭으로 정리한다. 번역으로 새로운 스펙을 추가하지 않는다.
- 쿠팡 API 호출이나 자동 등록은 하지 않는다. 사용자가 Wing 화면에 직접 옮겨 적을 수 있는 자료만 작성한다.

아래 순서로 누락 없이 출력:
1. 최종 판정: `등록 가능`, `확인 후 등록`, `등록 보류` 중 하나와 판정 근거
2. 근거 충돌표: 필드, 완료폴더 값, URL 값, 채택값, 상태(`확정/추정 금지/확인 필요`)
3. 노출상품명, 등록상품명, 브랜드, 제조사, 원산지. 각 값 옆에 근거 출처를 표시
4. 쿠팡 카테고리 후보 최대 3개와 displayCategoryCode 검수. 확인된 코드가 없으면 코드를 발명하지 않는다
5. 판매가, 재고, 배송비, 반품비, 교환비
6. 옵션별 표: 옵션명, 옵션값, 추가금, 재고, 판매상태, 근거
7. 카테고리 필수 구매옵션과 검색필터 속성 입력표: 입력값, 단위, 근거, 확인상태
8. 검색어 20개 이내와 제거해야 할 상표·금지·무관 검색어
9. 상품정보제공고시 입력표: 확인되지 않은 값은 `상세설명 참조`로 뭉뚱그리지 말고 구체적인 확인 항목 표시
10. 상세설명 HTML과 이미지 확인 항목
11. Wing 등록 화면에 위에서 아래로 입력하는 최종 체크리스트

각 항목은 바로 복사할 수 있는 실제 값부터 보여준다. `확정` 값과 `확인 필요` 값을 섞지 말고, 마지막에 사용자가 직접 확인해야 하는 항목만 짧게 다시 모아줘.
"""
        market_dir = product.folder / "market"
        market_dir.mkdir(parents=True, exist_ok=True)
        prompt_path = market_dir / "coupang_manual_chatgpt_prompt.txt"
        prompt_path.write_text(manual_prompt, encoding="utf-8")
        self._write_json_file(market_dir / "coupang_manual_evidence.json", evidence)
        attachments = [
            path for path in product.thumbnail_paths[:5]
            if self._is_valid_image_file(path)
        ]
        self.market_gpt_running = True
        self.statusBar().showMessage("쿠팡 수동등록 프롬프트를 ChatGPT로 보내는 중입니다.")

        def worker():
            try:
                self._ensure_cdp_browser()
                from playwright.sync_api import sync_playwright

                with sync_playwright() as playwright:
                    browser = playwright.chromium.connect_over_cdp("http://127.0.0.1:9222")
                    context = browser.contexts[0] if browser.contexts else browser.new_context()
                    page = context.new_page()
                    page.goto(CHATGPT_HOME_URL, wait_until="domcontentloaded", timeout=60000)
                    self._accept_chatgpt_cookies(page)
                    if self._page_needs_login(page):
                        page.bring_to_front()
                        raise RuntimeError("ChatGPT 로그인이 필요합니다. 로그인 후 다시 눌러주세요.")
                    self._send_prompt_to_gpt_page(
                        page,
                        manual_prompt,
                        attachment_paths=attachments,
                        image_mode=False,
                    )
                    page.bring_to_front()
                    self.automation_signals.status.emit(
                        f"쿠팡 수동등록용 프롬프트 전송 완료: {product.code} · {prompt_path}"
                    )
            except Exception as exc:
                self.automation_signals.status.emit(f"쿠팡 수동등록 GPT 전송 실패: {exc}")
            finally:
                self.market_gpt_running = False

        threading.Thread(target=worker, daemon=True).start()
